%
% Defined by branching HERE, at load time, rather than as
%   \newcommand{\bgen}[2]{\ifbg#1\else#2\fi}
% The second form leaves a live conditional inside the macro body, which breaks when
% the macro is expanded during a write into .toc or .aux. This form is fully
% expandable and is therefore safe inside \MakeUppercase and \addcontentsline.

  \newcommand{\bgen}[2]{#1}
\else
  \newcommand{\bgen}[2]{#2}
\fi

% siunitx: един формат за числата в целия документ, вместо решение на всяко
