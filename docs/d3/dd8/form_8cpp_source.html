94 std::string url_decode(std::string_view encoded)
95 {
96 std::string result;
97 result.reserve(encoded.size());
98
99 for (size_t i = 0; i < encoded.size(); ++i)
100 {
101 char c = encoded[i];
102 if (c == '+')
103 {
104 result += ' ';
105 }
106 else if (c == '%' && i + 2 < encoded.size())
107 {
108 int hi = hex_digit(encoded[i + 1]);
109 int lo = hex_digit(encoded[i + 2]);
110 if (hi >= 0 && lo >= 0)
111 {
112 result += static_cast<char>((hi << 4) | lo);
113 i += 2;
114 }
115 else
116 {
117 result += c;
118 }
119 }
120 else
121 {
122 result += c;
123 }
124 }
125
126 return result;
127 }

129 std::string url_encode(std::string_view decoded)
130 {
131 std::ostringstream oss;
132 oss << std::hex << std::uppercase;
133
134 for (char c : decoded)
135 {
136 if (std::isalnum(static_cast<unsigned char>(c)) || c == '-' || c == '_' || c == '.' || c == '~')
137 {
138 oss << c;
139 }
140 else if (c == ' ')
141 {
142 oss << '+';
143 }
144 else
145 {
146 oss << '%' << std::setw(2) << std::setfill('0') << static_cast<int>(static_cast<unsigned char>(c));
147 }
148 }
149
150 return oss.str();

157 expected<FormData, Error> parse_urlencoded(std::string_view body)
158 {
159 FormData data;
160
161 if (body.empty())
162 {
163 return data;
164 }
165
166 size_t start = 0;
167 while (start < body.size())
168 {
169 // Find end of this pair
170 size_t end = body.find('&', start);
171 if (end == std::string_view::npos)
172 {
173 end = body.size();
174 }
175
176 auto pair = body.substr(start, end - start);
177
178 // Find = separator
179 auto eq = pair.find('=');
180 std::string name, value;
181
182 if (eq == std::string_view::npos)
183 {
184 name = url_decode(pair);
185 }
186 else
187 {
188 name = url_decode(pair.substr(0, eq));
189 value = url_decode(pair.substr(eq + 1));
190 }
191
192 if (!name.empty())
193 {
194 data.add(std::move(name), std::move(value));
195 }
196
197 start = end + 1;
198 }
199
200 return data;
201 }

258 expected<FormData, Error> parse_multipart(std::string_view body, std::string_view boundary)
259 {
260 FormData data;
261
262 if (body.empty() || boundary.empty())
263 {
264 return data;
265 }
266
267 std::string delimiter = "--" + std::string(boundary);
268
269 size_t pos = body.find(delimiter);
270 if (pos == std::string_view::npos)
271 {
272 return unexpected(Error::http(HttpError::BadRequest, "Multipart boundary not found"));
273 }
274
275 pos += delimiter.size();
276
277 while (pos < body.size())
278 {
279 // Skip CRLF after boundary
280 if (body.substr(pos, 2) == "\r\n")
281 {
282 pos += 2;
283 }
284 else if (body.substr(pos, 2) == "--")
285 {
286 // End of multipart
287 break;
288 }
289
290 // Find end of headers (double CRLF)
291 auto headers_end = body.find("\r\n\r\n", pos);
292 if (headers_end == std::string_view::npos)
293 {
294 break;
295 }
296
297 // Parse headers
298 auto headers_section = body.substr(pos, headers_end - pos);
299 std::string content_disposition;
300 std::string content_type;
301
302 size_t line_start = 0;
303 while (line_start < headers_section.size())
304 {
305 auto line_end = headers_section.find("\r\n", line_start);
306 if (line_end == std::string_view::npos)
307 {
308 line_end = headers_section.size();
309 }
310
311 auto line = headers_section.substr(line_start, line_end - line_start);
312 auto colon = line.find(':');
313 if (colon != std::string_view::npos)
314 {
315 auto header_name = trim(line.substr(0, colon));
316 auto header_value = trim(line.substr(colon + 1));
317
318 // Case-insensitive comparison
319 std::string name_lower(header_name);
320 std::transform(name_lower.begin(), name_lower.end(), name_lower.begin(),
321 [](unsigned char c) { return std::tolower(c); });
322
323 if (name_lower == "content-disposition")
324 {
325 content_disposition = std::string(header_value);
326 }
327 else if (name_lower == "content-type")
328 {
329 content_type = std::string(header_value);
330 }
331 }
332
333 line_start = line_end + 2;
334 }
335
336 // Parse Content-Disposition
337 auto [name, filename] = parse_content_disposition(content_disposition);
338
339 // Find content (until next boundary)
340 pos = headers_end + 4; // Skip \r\n\r\n
341 auto next_boundary = body.find(delimiter, pos);
342 if (next_boundary == std::string_view::npos)
343 {
344 break;
345 }
346
347 // Content is everything until \r\n before boundary
348 size_t content_end = next_boundary;
349 if (content_end >= 2 && body.substr(content_end - 2, 2) == "\r\n")
350 {
351 content_end -= 2;
352 }
353
354 std::string value(body.substr(pos, content_end - pos));
355
356 // Create field
357 FormField field;
358 field.name = std::move(name);
359 field.value = std::move(value);
360 field.filename = std::move(filename);
361 field.content_type = std::move(content_type);
362 field.is_file = !field.filename.empty();
363
364 if (!field.name.empty())
365 {
366 data.add(std::move(field));
367 }
368
369 pos = next_boundary + delimiter.size();
370 }
371
372 return data;
373 }

375 std::optional<std::string> extract_boundary(std::string_view content_type)
376 {
377 // Look for boundary=...
378 auto boundary_pos = content_type.find("boundary=");
379 if (boundary_pos == std::string_view::npos)
380 {
381 return std::nullopt;
382 }
383
384 boundary_pos += 9; // Skip "boundary="
385
386 // Check for quoted boundary
387 if (boundary_pos < content_type.size() && content_type[boundary_pos] == '"')
388 {
389 ++boundary_pos;
390 auto end = content_type.find('"', boundary_pos);
391 if (end == std::string_view::npos)
392 {
393 return std::nullopt;
394 }
395 return std::string(content_type.substr(boundary_pos, end - boundary_pos));
396 }
397
398 // Unquoted boundary (ends at ; or end of string)
399 auto end = content_type.find(';', boundary_pos);
400 if (end == std::string_view::npos)
401 {
402 end = content_type.size();
403 }
404
405 auto boundary = trim(content_type.substr(boundary_pos, end - boundary_pos));
406 return std::string(boundary);
407 }

409 expected<FormData, Error> parse(const Request& req)
410 {
411 auto content_type = req.content_type();
412 if (!content_type)
413 {
414 return unexpected(Error::http(HttpError::BadRequest, "Missing Content-Type header"));
415 }
416
417 if (content_type->starts_with("application/x-www-form-urlencoded"))
418 {
419 return parse_urlencoded(req.body());
420 }
421
422 if (content_type->starts_with("multipart/form-data"))
423 {
424 auto boundary = extract_boundary(*content_type);
425 if (!boundary)
426 {
427 return unexpected(Error::http(HttpError::BadRequest, "Missing boundary in multipart Content-Type"));
428 }
429 return parse_multipart(req.body(), *boundary);
430 }
431
432 return unexpected(Error::http(HttpError::UnsupportedMediaType, "Unsupported form content type"));
433 }
