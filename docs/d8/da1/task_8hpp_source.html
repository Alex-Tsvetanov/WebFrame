245 std::mutex mtx;
246 std::condition_variable cv;
247 bool finished = false;

249 // Install a completion callback that signals the CV
250 handle_.promise().on_complete_ = [&]
251 {
252 {
253 std::lock_guard<std::mutex> lock(mtx);
254 finished = true;
255 }

257 };

260 handle_.resume();

262 // Block until FinalAwaiter fires the callback
263 {
264 std::unique_lock<std::mutex> lock(mtx);
265 cv.wait(lock, [&] { return finished; });
266 }
267
268 // Clear the callback to avoid dangling references
269 handle_.promise().on_complete_ = nullptr;

271 if constexpr (std::is_void_v<T>)
272 {
273 handle_.promise().result();
