55 bool is_websocket_upgrade(const Request& req)
56 {
57 // Check for required headers
58 auto connection = req.header("Connection");
59 auto upgrade = req.header("Upgrade");
60 auto ws_key = req.header("Sec-WebSocket-Key");
61 auto ws_version = req.header("Sec-WebSocket-Version");
62
63 if (!connection || !upgrade || !ws_key || !ws_version)
64 {
65 return false;
66 }
67
68 // Check header values (case-insensitive for Connection and Upgrade)
69 bool has_upgrade = false;
70 std::string conn_lower(*connection);
71 for (auto& c : conn_lower) c = static_cast<char>(std::tolower(c));
72 has_upgrade = conn_lower.find("upgrade") != std::string::npos;
73
74 std::string upgrade_lower(*upgrade);
75 for (auto& c : upgrade_lower) c = static_cast<char>(std::tolower(c));
76 bool is_websocket = upgrade_lower == "websocket";
77
78 // Version must be 13
79 bool version_ok = *ws_version == "13";
80
81 return has_upgrade && is_websocket && version_ok;
82 }

142 // Frame header structure:
143 // Byte 0: FIN (1 bit) + RSV1-3 (3 bits) + Opcode (4 bits)
144 // Byte 1: MASK (1 bit) + Payload length (7 bits)
145 // If payload length == 126: next 2 bytes are actual length (16-bit)
146 // If payload length == 127: next 8 bytes are actual length (64-bit)
147 // If MASK bit set: next 4 bytes are masking key
148
149 struct FrameHeader
150 {
151 bool fin = true;
152 bool rsv1 = false;
153 bool rsv2 = false;
154 bool rsv3 = false;
155 WebSocketOpcode opcode = WebSocketOpcode::Text;
156 bool masked = false;
157 uint64_t payload_length = 0;
158 std::array<uint8_t, 4> mask_key{};
159 };
160
161 // Parse frame header from buffer
162 // Returns number of bytes consumed, or 0 if not enough data
163 size_t parse_frame_header(const uint8_t* data, size_t len, FrameHeader& header)
164 {
165 if (len < 2) return 0;
166
167 header.fin = (data[0] & 0x80) != 0;
168 header.rsv1 = (data[0] & 0x40) != 0;
169 header.rsv2 = (data[0] & 0x20) != 0;
170 header.rsv3 = (data[0] & 0x10) != 0;
171 header.opcode = static_cast<WebSocketOpcode>(data[0] & 0x0F);
172 header.masked = (data[1] & 0x80) != 0;
173
174 uint8_t len_byte = data[1] & 0x7F;
175 size_t header_size = 2;
176
177 if (len_byte <= 125)
178 {
179 header.payload_length = len_byte;
180 }
181 else if (len_byte == 126)
182 {
183 if (len < 4) return 0;
184 header.payload_length = (static_cast<uint64_t>(data[2]) << 8) | data[3];
185 header_size = 4;
186 }
187 else
188 { // 127
189 if (len < 10) return 0;
190 header.payload_length = 0;
191 for (int i = 0; i < 8; ++i)
192 {
193 header.payload_length = (header.payload_length << 8) | data[2 + i];
194 }
195 header_size = 10;
196 }
197
198 if (header.masked)
199 {
200 if (len < header_size + 4) return 0;
201 std::memcpy(header.mask_key.data(), data + header_size, 4);
202 header_size += 4;
203 }
204
205 return header_size;
206 }

208 // Serialize frame header
209 std::vector<uint8_t> serialize_frame_header(const FrameHeader& header)
210 {
211 std::vector<uint8_t> result;
212
213 auto byte0 = static_cast<uint8_t>(header.opcode);
214 if (header.fin) byte0 |= 0x80;
215 if (header.rsv1) byte0 |= 0x40;
216 if (header.rsv2) byte0 |= 0x20;
217 if (header.rsv3) byte0 |= 0x10;
218 result.push_back(byte0);
219
220 uint8_t byte1 = header.masked ? 0x80 : 0x00;
221 if (header.payload_length <= 125)
222 {
223 byte1 |= static_cast<uint8_t>(header.payload_length);
224 result.push_back(byte1);
225 }
226 else if (header.payload_length <= 0xFFFF)
227 {
228 byte1 |= 126;
229 result.push_back(byte1);
230 result.push_back(static_cast<uint8_t>(header.payload_length >> 8));
231 result.push_back(static_cast<uint8_t>(header.payload_length & 0xFF));
232 }
233 else
234 {
235 byte1 |= 127;
236 result.push_back(byte1);
237 for (int i = 7; i >= 0; --i)
238 {
239 result.push_back(static_cast<uint8_t>(header.payload_length >> (i * 8)));
240 }
241 }
242
243 if (header.masked)
244 {
245 result.insert(result.end(), header.mask_key.begin(), header.mask_key.end());
246 }
247
248 return result;
249 }
250
251 // Apply/remove XOR mask to payload
252 void apply_mask(uint8_t* data, size_t len, const std::array<uint8_t, 4>& mask)
253 {
254 for (size_t i = 0; i < len; ++i)
255 {
256 data[i] ^= mask[i % 4]; // NOLINT(cppcoreguidelines-pro-bounds-constant-array-index)
257 }
258 }

281 Task<expected<WebSocketMessage, Error>> receive() override
282 {
283 while (is_open_)
284 {
285 // Try to parse a frame from existing buffer
286 if (!read_buffer_.empty())
287 {
288 FrameHeader header;
289 size_t header_size = parse_frame_header(read_buffer_.data(), read_buffer_.size(), header);
290
291 if (header_size > 0 && read_buffer_.size() >= header_size + header.payload_length)
292 {
293 // We have a complete frame
294 std::vector<uint8_t> payload(read_buffer_.begin() + header_size,
295 read_buffer_.begin() + header_size + header.payload_length);
296
297 // Remove frame from buffer
298 read_buffer_.erase(read_buffer_.begin(),
299 read_buffer_.begin() + header_size + header.payload_length);
300
301 // Unmask if needed (client frames must be masked)
302 if (header.masked)
303 {
304 apply_mask(payload.data(), payload.size(), header.mask_key);
305 }
306
307 // Handle control frames immediately
308 if (header.opcode == WebSocketOpcode::Ping)
309 {
310 co_await pong(payload);
311 continue;
312 }
313
314 if (header.opcode == WebSocketOpcode::Pong)
315 {
316 // Ignore pongs for now
317 continue;
318 }
319
320 if (header.opcode == WebSocketOpcode::Close)
321 {
322 is_open_ = false;
323 // Echo close frame
324 WebSocketCloseCode code = WebSocketCloseCode::Normal;
325 std::string_view reason;
326 if (payload.size() >= 2)
327 {
328 code = static_cast<WebSocketCloseCode>((payload[0] << 8) | payload[1]);
329 if (payload.size() > 2)
330 {
331 reason = std::string_view(
332 reinterpret_cast<const char*>(
333 payload.data() + 2), // NOLINT(cppcoreguidelines-pro-type-reinterpret-cast)
334 payload.size() - 2);
335 }
336 }
337 co_await send_close_frame(code, reason);
338 co_return WebSocketMessage{.opcode = WebSocketOpcode::Close, .data = std::move(payload)};
339 }
340
341 // Handle data frames
342 if (header.opcode == WebSocketOpcode::Continuation)
343 {
344 // Continuation of fragmented message
345 fragment_buffer_.insert(fragment_buffer_.end(), payload.begin(), payload.end());
346
347 if (fragment_buffer_.size() > max_message_size_)
348 {
349 co_await close(WebSocketCloseCode::MessageTooBig, "Message too large");
350 co_return unexpected(Error::io(IoError::Unknown, "Message too large"));
351 }
352
353 if (header.fin)
354 {
355 // Final fragment
356 WebSocketMessage msg{.opcode = fragment_opcode_, .data = std::move(fragment_buffer_)};
357 fragment_buffer_.clear();
358 co_return msg;
359 }
360 }
361 else
362 {
363 // New message
364 if (!fragment_buffer_.empty())
365 {
366 // Protocol error: new message while fragmented message incomplete
367 co_await close(WebSocketCloseCode::ProtocolError, "Incomplete fragmented message");
368 co_return unexpected(Error::io(IoError::Unknown, "Protocol error"));
369 }
370
371 if (header.fin)
372 {
373 // Complete message in single frame
374 co_return WebSocketMessage{.opcode = header.opcode, .data = std::move(payload)};
375 }
376 else
377 {
378 // Start of fragmented message
379 fragment_opcode_ = header.opcode;
380 fragment_buffer_ = std::move(payload);
381 }
382 }
383
384 continue;
385 }
386 }
387
388 // Need more data
389 std::array<uint8_t, 8192> buf;
390 auto result = co_await conn_->async_read(buf.data(), buf.size());
391 if (!result)
392 {
393 is_open_ = false;
394 co_return unexpected(result.error());
395 }
396 if (*result == 0)
397 {
398 is_open_ = false;
399 co_return unexpected(Error::io(IoError::ConnectionReset, "Connection closed"));
400 }
401
402 read_buffer_.insert(read_buffer_.end(), buf.begin(), buf.begin() + *result);
403 }
404
405 co_return unexpected(Error::io(IoError::ConnectionReset, "Connection closed"));
406 }

448 Task<expected<void, Error>> send_frame(WebSocketOpcode opcode, const uint8_t* data, size_t len)
449 {
450 FrameHeader header;
451 header.fin = true;
452 header.opcode = opcode;
453 header.masked = false; // Server frames are not masked
454 header.payload_length = len;
455
456 auto header_bytes = serialize_frame_header(header);
457
458 // Send header
459 auto header_result = co_await conn_->async_write_all(header_bytes.data(), header_bytes.size());
460 if (!header_result) co_return unexpected(header_result.error());
461
462 // Send payload
463 if (len > 0)
464 {
465 auto payload_result = co_await conn_->async_write_all(data, len);
466 if (!payload_result) co_return unexpected(payload_result.error());
467 }
468
469 co_return expected<void, Error>{};
470 }

477 payload.insert(payload.end(), reason.begin(), reason.end());

479 co_return co_await send_frame(WebSocketOpcode::Close, payload.data(), payload.size());

489 {

498 // Serialize response

501 resp_str += std::to_string(response.status());
502 resp_str += " ";
503 resp_str += response.status_text();
504 resp_str += "\r\n";
505

508 resp_str += name;
509 resp_str += ": ";

511 resp_str += "\r\n";
512 }

514

516 if (!result)

518 co_return unexpected(result.error());
519 }
