107 CookieJar CookieJar::parse(std::string_view header)
108 {
109 CookieJar jar;
110
111 size_t start = 0;
112 while (start < header.size())
113 {
114 // Find end of this cookie (semicolon or end)
115 size_t end = header.find(';', start);
116 if (end == std::string_view::npos)
117 {
118 end = header.size();
119 }
120
121 auto pair = trim(header.substr(start, end - start));
122
123 // Find = separator
124 auto eq = pair.find('=');
125 if (eq != std::string_view::npos)
126 {
127 auto name = trim(pair.substr(0, eq));
128 auto value = trim(pair.substr(eq + 1));
129
130 // Remove quotes if present
131 if (value.size() >= 2 && value.front() == '"' && value.back() == '"')
132 {
133 value = value.substr(1, value.size() - 2);
134 }
135
136 if (!name.empty())
137 {
138 jar.cookies_[std::string(name)] = std::string(value);
139 }
140 }
141
142 start = end + 1;
143 }
